foreach(backend IN ITEMS Composia Direct2D Gdi)
    add_executable(bench-${backend} ${backend}.cpp ${COMPOSIA_MANIFEST})
    target_compile_features(bench-${backend} PRIVATE cxx_std_23)
    target_compile_options(bench-${backend} PRIVATE /EHsc /W4 $<$<BOOL:${COMPOSIA_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_definitions(bench-${backend} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(bench-${backend} PRIVATE WIL::WIL psapi)
endforeach()
target_link_libraries(bench-Composia PRIVATE Composia::UI)
target_link_libraries(bench-Direct2D PRIVATE d2d1 d3d11 dxgi)
target_link_libraries(bench-Gdi PRIVATE user32 gdi32)
add_custom_target(composia-benchmarks DEPENDS bench-Composia bench-Direct2D bench-Gdi)
